-- passos 4 a 6: a trigger do estoque
DROP TRIGGER IF EXISTS tg_item_baixa_estoque;
DELIMITER //
CREATE TRIGGER tg_item_baixa_estoque
AFTER INSERT ON item_pedido
FOR EACH ROW
BEGIN
    UPDATE estoque
    SET quantidade = quantidade - NEW.quantidade
    WHERE produto_codigo = NEW.produto_codigo
      AND unidade_codigo = (SELECT unidade_codigo FROM pedido WHERE codigo = NEW.pedido_codigo);
END //
DELIMITER ;

Parte 2

-- passos 7 a 9: o log e a trigger de auditoria
CREATE TABLE IF NOT EXISTS log_preco_produto (
    id INT AUTO_INCREMENT PRIMARY KEY,
    produto_codigo INT NOT NULL,
    preco_antigo DECIMAL(10, 2) NOT NULL,
    preco_novo DECIMAL(10, 2) NOT NULL,
    alterado_em DATETIME DEFAULT CURRENT_TIMESTAMP,
    usuario VARCHAR(100),
    FOREIGN KEY (produto_codigo) REFERENCES produto(codigo)
);

DROP TRIGGER IF EXISTS tg_produto_preco;
DELIMITER //
CREATE TRIGGER tg_produto_preco
AFTER UPDATE ON produto
FOR EACH ROW
BEGIN
    IF NEW.preco_tabela <> OLD.preco_tabela THEN
        INSERT INTO log_preco_produto (produto_codigo, preco_antigo, preco_novo, usuario)
        VALUES (NEW.produto_codigo, OLD.preco_tabela, NEW.preco_tabela, USER());
    END IF;
END //
DELIMITER ;
